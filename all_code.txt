class DeviceController
{
    private DeviceRepository $deviceRepo;

    public function __construct(DeviceRepository $deviceRepo)
    {
        $this->deviceRepo = $deviceRepo;
    }

    public function index(): void
    {
        $devices = $this->deviceRepo->getAllDevices();
        $deviceTypes = $this->deviceRepo->getDeviceTypes();
        $rooms = $this->deviceRepo->getRooms();
        $bookings = $this->deviceRepo->getBookings();

        $message = $_SESSION['device_message'] ?? '';
        unset($_SESSION['device_message']);

        $page = 'devices';

        require __DIR__ . '/../View/staff/themthietbitv4.php';
    }

    public function handleRequest(): void
    {
        if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
            $this->index();
            return;
        }

        $action = $_POST['action'] ?? '';

        switch ($action) {
            case 'add':
                $this->addDevice();
                break;

            case 'edit':
                $this->editDevice();
                break;

            case 'delete':
                $this->deleteDevice();
                break;

            case 'booking':
                $this->handleBooking();
                break;

            default:
                $this->index();
                break;
        }
    }

    private function addDevice(): void
    {
        $deviceCode = trim($_POST['device_code'] ?? '');
        $deviceName = trim($_POST['device_name'] ?? '');
        $typeId = (int)($_POST['type_id'] ?? 0);
        $roomId = (int)($_POST['room_id'] ?? 0);
        $status = $_POST['status'] ?? 'active';

        if (
            $deviceCode === '' ||
            $deviceName === '' ||
            $typeId <= 0 ||
            $roomId <= 0
        ) {
            $_SESSION['device_message'] = 'Vui lÃ²ng nháº­p Ä‘áº§y Ä‘á»§ thÃ´ng tin thiáº¿t bá»‹.';
            $this->redirect();
        }

        if (!in_array($status, ['active', 'broken', 'maintenance'], true)) {
            $status = 'active';
        }

        try {
            $this->deviceRepo->addDevice(
                $deviceCode,
                $deviceName,
                $typeId,
                $roomId,
                $status
            );

            $_SESSION['device_message'] = 'ThÃªm thiáº¿t bá»‹ thÃ nh cÃ´ng.';
        } catch (PDOException $e) {
            $_SESSION['device_message'] = 'KhÃ´ng thá»ƒ thÃªm thiáº¿t bá»‹. MÃ£ thiáº¿t bá»‹ cÃ³ thá»ƒ Ä‘Ã£ tá»“n táº¡i.';
        }

        $this->redirect();
    }

    private function editDevice(): void
    {
        $id = (int)($_POST['id'] ?? 0);
        $deviceCode = trim($_POST['device_code'] ?? '');
        $deviceName = trim($_POST['device_name'] ?? '');
        $typeId = (int)($_POST['type_id'] ?? 0);
        $roomId = (int)($_POST['room_id'] ?? 0);
        $status = $_POST['status'] ?? 'active';

        if (
            $id <= 0 ||
            $deviceCode === '' ||
            $deviceName === '' ||
            $typeId <= 0 ||
            $roomId <= 0
        ) {
            $_SESSION['device_message'] = 'ThÃ´ng tin thiáº¿t bá»‹ khÃ´ng há»£p lá»‡.';
            $this->redirect();
        }

        if (!in_array($status, ['active', 'broken', 'maintenance'], true)) {
            $status = 'active';
        }

        try {
            $this->deviceRepo->updateDevice(
                $id,
                $deviceCode,
                $deviceName,
                $typeId,
                $roomId,
                $status
            );

            $_SESSION['device_message'] = 'Cáº­p nháº­t thiáº¿t bá»‹ thÃ nh cÃ´ng.';
        } catch (PDOException $e) {
            $_SESSION['device_message'] = 'KhÃ´ng thá»ƒ cáº­p nháº­t thiáº¿t bá»‹. MÃ£ thiáº¿t bá»‹ cÃ³ thá»ƒ Ä‘Ã£ tá»“n táº¡i.';
        }

        $this->redirect();
    }

    private function deleteDevice(): void
    {
        $id = (int)($_POST['id'] ?? 0);

        if ($id <= 0) {
            $_SESSION['device_message'] = 'Thiáº¿t bá»‹ khÃ´ng há»£p lá»‡.';
            $this->redirect();
        }

        try {
            $this->deviceRepo->deleteDevice($id);

        } catch (PDOException $e) {
            $_SESSION['device_message'] = 'KhÃ´ng thá»ƒ xÃ³a thiáº¿t bá»‹.';
        }

        $this->redirect();
    }

    private function handleBooking(): void
    {
        $id = (int)($_POST['id'] ?? 0);
        $status = $_POST['status'] ?? '';

        if (
            $id <= 0 ||
            !in_array($status, ['approved', 'rejected'], true)
        ) {
            $_SESSION['device_message'] = 'YÃªu cáº§u phÃ²ng khÃ´ng há»£p lá»‡.';
            $this->redirect();
        }

        $booking = $this->deviceRepo->getBookingById($id);

        if (!$booking) {
            $_SESSION['device_message'] = 'KhÃ´ng tÃ¬m tháº¥y yÃªu cáº§u Ä‘áº·t phÃ²ng.';
            $this->redirect();
        }

        if ($booking['status'] !== 'pending') {
            $_SESSION['device_message'] = 'YÃªu cáº§u nÃ y Ä‘Ã£ Ä‘Æ°á»£c xá»­ lÃ½.';
            $this->redirect();
        }

        if ($status === 'approved') {
            $conflict = $this->deviceRepo->checkBookingConflict(
                (int)$booking['room_id'],
                $booking['start_time'],
                $booking['end_time'],
                (int)$booking['id']
            );

            if ($conflict) {
                $_SESSION['device_message'] = 'KhÃ´ng thá»ƒ duyá»‡t. PhÃ²ng Ä‘Ã£ cÃ³ lá»‹ch Ä‘Æ°á»£c duyá»‡t trong khoáº£ng thá»i gian nÃ y.';
                $this->redirect();
            }
        }

        $this->deviceRepo->updateBookingStatus($id, $status);

        if ($status === 'approved') {
            $_SESSION['device_message'] = 'ÄÃ£ duyá»‡t yÃªu cáº§u Ä‘áº·t phÃ²ng.';
        } else {
            $_SESSION['device_message'] = 'ÄÃ£ tá»« chá»‘i yÃªu cáº§u Ä‘áº·t phÃ²ng.';
        }

        $this->redirect();
    }

    private function redirect(): void
    {
        header('Location: index.php?page=devices');
        exit;
    }
}

    /*
     * =====================================================
     * TV3 - Xá»¬ LÃ Äáº¶T PHÃ’NG
     * =====================================================
     */
    private function handleBooking(int $userId): array
    {
        $errors = [];
        $success = '';

        $roomId = trim(
            (string) ($_POST['room_id'] ?? '')
        );

        $date = trim(
            (string) ($_POST['date'] ?? '')
        );

        $start = trim(
            (string) ($_POST['start'] ?? '')
        );

        $end = trim(
            (string) ($_POST['end'] ?? '')
        );

        $purpose = trim(
            (string) ($_POST['purpose'] ?? '')
        );


        /*
         * VALIDATE
         */

        if ($roomId === '') {
            $errors['room_id']
                = 'Vui lÃ²ng chá»n phÃ²ng.';
        }

        if ($date === '') {
            $errors['date']
                = 'Vui lÃ²ng chá»n ngÃ y.';
        }

        if ($start === '') {
            $errors['start']
                = 'Vui lÃ²ng chá»n giá» báº¯t Ä‘áº§u.';
        }

        if ($end === '') {
            $errors['end']
                = 'Vui lÃ²ng chá»n giá» káº¿t thÃºc.';
        }

        if (
            $start !== ''
            && $end !== ''
            && $end <= $start
        ) {
            $errors['end']
                = 'Giá» káº¿t thÃºc pháº£i lá»›n hÆ¡n giá» báº¯t Ä‘áº§u.';
        }

        if ($purpose === '') {
            $errors['purpose']
                = 'Vui lÃ²ng nháº­p má»¥c Ä‘Ã­ch.';
        }


        if (!empty($errors)) {
            return [$errors, $success];
        }


        /*
         * KIá»‚M TRA PHÃ’NG
         */

        $room = $this->roomRepo->findBookableRoom(
            (int) $roomId
        );

        if (!$room) {

            $errors['room_id']
                = 'PhÃ²ng khÃ´ng tá»“n táº¡i hoáº·c Ä‘ang báº£o trÃ¬.';

            return [$errors, $success];
        }


        /*
         * Táº O DATETIME
         */

        $startTime =
            $date . ' ' . $start . ':00';

        $endTime =
            $date . ' ' . $end . ':00';


        /*
         * KIá»‚M TRA TRÃ™NG Lá»ŠCH
         *
         * TV3
         */

        if (
            $this->roomRepo->hasBookingConflict(
                (int) $roomId,
                $startTime,
                $endTime
            )
        ) {

            $errors['room_id']
                = 'PhÃ²ng Ä‘Ã£ Ä‘Æ°á»£c Ä‘áº·t trong thá»i gian nÃ y.';

            return [$errors, $success];
        }


        /*
         * INSERT BOOKING
         */

        $this->roomRepo->createBooking(
            $userId,
            (int) $roomId,
            $startTime,
            $endTime,
            $purpose
        );

        $success = 'Äáº·t phÃ²ng thÃ nh cÃ´ng.';

        return [$errors, $success];
    }


    /*
     * =====================================================
     * TV3 - Há»¦Y BOOKING
     * =====================================================
     */
    private function handleCancelBooking(
        int $userId
    ): array {

        $errors = [];
        $success = '';

        $bookingId =
            (int) ($_POST['booking_id'] ?? 0);


        if ($bookingId <= 0) {

            $errors['cancel']
                = 'YÃªu cáº§u Ä‘áº·t phÃ²ng khÃ´ng há»£p lá»‡.';

            return [$errors, $success];
        }


        /*
         * CHá»ˆ ÄÆ¯á»¢C Há»¦Y BOOKING Cá»¦A CHÃNH MÃŒNH
         */

        $booking =
            $this->roomRepo->findBookingForUser(
                $bookingId,
                $userId
            );


        if (!$booking) {

            $errors['cancel']
                = 'KhÃ´ng tÃ¬m tháº¥y yÃªu cáº§u Ä‘áº·t phÃ²ng cá»§a báº¡n.';

            return [$errors, $success];
        }


        /*
         * KIá»‚M TRA CÃ“ THá»‚ Há»¦Y KHÃ”NG
         */

        $canCancel =
            $booking['status'] !== 'cancelled'
            &&
            (
                $booking['status'] === 'pending'
                ||
                $booking['start_time']
                    > date('Y-m-d H:i:s')
            );


        if (!$canCancel) {

            $errors['cancel']
                = 'KhÃ´ng thá»ƒ há»§y yÃªu cáº§u nÃ y.';

            return [$errors, $success];
        }


        $this->roomRepo->cancelBooking(
            $bookingId,
            $userId
        );

        $success =
            'Há»§y yÃªu cáº§u Ä‘áº·t phÃ²ng thÃ nh cÃ´ng.';

        return [$errors, $success];
    }


    /*
     * =====================================================
     * TV2 - BÃO Há»ŽNG THIáº¾T Bá»Š
     * =====================================================
     */
    private function handleReport(

    ): array {

        $errors = [];
        $success = '';

        $deviceId =
            trim(
                (string) ($_POST['device_id'] ?? '')
            );

        $description =

        $success =
            'BÃ¡o há»ng thiáº¿t bá»‹ thÃ nh cÃ´ng.';

<?php

namespace App\Repository;

use PDO;

class DeviceRepository
{
    private PDO $pdo;

    public function __construct(PDO $pdo)
    {
        $this->pdo = $pdo;
    }

    public function getAllDevices(): array
    {
        $sql = "SELECT
                    d.id,
                    d.device_code,
                    d.device_name,
                    d.room_id,
                    d.type_id,
                    d.status,
                    dt.name AS type_name,
                    r.room_code,
                    r.room_name
                FROM devices d
                INNER JOIN device_types dt ON d.type_id = dt.id
                INNER JOIN rooms r ON d.room_id = r.id
                ORDER BY d.id DESC";

        return $this->pdo->query($sql)->fetchAll(PDO::FETCH_ASSOC);
    }

    public function getDeviceTypes(): array

function findName(list,id){

    const item = list.find(
        x => Number(x.id) === Number(id)
    );

    if(!item){
        return "";
    }

    return item.name ||
           item.full_name ||
           item.device_name ||
           item.room_name ||
           item.room_code ||
           "";
}

function dateKey(value){

    const match = String(value).match(
        /(\d{2})\/(\d{2})\/(\d{4})/
    );

    return match
        ? `${match[3]}-${match[2]}-${match[1]}`
        : "";
}

function taoLichCalendar(){

    const grid = document.getElementById("calendarGrid");
    const title = document.getElementById("calendarTitle");

    if(!grid || !title){
        return;
    }

    const year = calendarDate.getFullYear();
    const month = calendarDate.getMonth();

    title.textContent =
        `ThÃ¡ng ${month + 1} ${year}`;

    const names = [
        "T2",
        "T3",
        "T4",
        "T5",
        "T6",
        "T7",
        "CN"
    ];

    let html = names
        .map(
            x => `<div class="calendar-weekday">${x}</div>`
        )
        .join("");

    let start =
        new Date(year,month,1).getDay();

    start =
        start === 0
        ? 6
        : start - 1;

    const days =
        new Date(year,month + 1,0).getDate();

    const prevDays =
        new Date(year,month,0).getDate();

    const total =
        Math.ceil((start + days) / 7) * 7;

    const now = new Date();

    const today =
        `${now.getFullYear()}-${String(
            now.getMonth()+1
        ).padStart(2,"0")}-${String(
            now.getDate()
        ).padStart(2,"0")}`;

    for(let i = 0; i < total; i++){

        let day;
        let cell;

        if(i < start){

            day =
                prevDays - start + i + 1;

            cell =
                new Date(year,month - 1,day);

        }else if(i < start + days){

            day =
                i - start + 1;

            cell =
                new Date(year,month,day);

        }else{

            day =
                i - start - days + 1;

            cell =
                new Date(year,month + 1,day);
        }

        const key =
            `${cell.getFullYear()}-${String(
                cell.getMonth()+1
            ).padStart(2,"0")}-${String(
                cell.getDate()
            ).padStart(2,"0")}`;

        let events = [];

        bookingCalendarData.forEach(item => {

            if(
                item.status !== "rejected" &&
                dateKey(item.start_time) === key
            ){

                events.push({

                    type:"room",

                    text:
                        `${item.start_time.split(" ")[0]} ${
                            findName(calendarRooms,item.room_id)
                        }`,

                    detail:
                        `${findName(calendarUsers,item.user_id)} â€¢ ${
                            item.start_time
                        } - ${
                            item.end_time
                        }`
                });
            }
        });

        borrowingCalendarData.forEach(item => {

            if(
                item.status !== "rejected" &&
                dateKey(item.start_time) === key
            ){

                const device =
                    calendarDevices.find(
                        x =>
                            Number(x.id) ===
                            Number(item.device_id)
                    );

                const deviceName =
                    device
                    ? (
                        device.device_name ||
                        device.name ||
                        ""
                    )
                    : "";

                events.push({

                    type:"borrow",

                    text:
                        `${item.start_time.split(" ")[0]} ${deviceName}`,

                    detail:
                        `${findName(calendarUsers,item.user_id)} â€¢ ${
                            item.start_time
                        } - ${
                            item.end_time
                        }`
                });
            }
        });

        events.sort(
            (a,b) =>
                a.text.localeCompare(
                    b.text,
                    "vi"
                )
        );

        const visible =
            events.slice(0,3);

        const hidden =
            events.length - visible.length;

        const other =
            cell.getMonth() !== month
            ? "other-month"
            : "";

        const current =
            key === today
            ? "today"
            : "";

        html +=
            `<div class="calendar-day ${other} ${current}">
                <div class="calendar-date">${day}</div>
                <div class="calendar-events">`;

        visible.forEach(event => {

            const detail =
                event.detail.replace(
                    /"/g,
                    "&quot;"
                );

            html +=
                `<div
                    class="calendar-event ${event.type}"
                    title="${detail}"
                >
                    ${event.text}
                </div>`;
        });

        if(hidden > 0){

            html +=
                `<div class="calendar-more">
                    +${hidden} lá»‹ch khÃ¡c
                </div>`;
        }

        html +=
            `</div></div>`;
    }

    grid.innerHTML = html;
}

$pdo = Database::connection();

switch ($route) {

    case 'login':
        require_once __DIR__ . '/../src/View/auth/login.php';
        break;

    case 'register':
        require_once __DIR__ . '/../src/View/auth/register.php';

    case 'users':

        $userController = new UserController($userRepo);
        $userController->index();

    case 'rooms':
        require_once __DIR__ . '/../src/View/student/roombooking.php';

    case 'bookings':

        $bookingController = new RoomBookingController($bookingRepo);
