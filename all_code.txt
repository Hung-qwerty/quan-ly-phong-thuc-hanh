    private function handleBooking(): void

                'error'

                (int)$booking['room_id'],
                $booking['start_time'],
                $booking['end_time'],
                (int)$booking['id']
            );

            if ($conflict) {
                $this->setMessage(
                    'KhÃ´ng thá»ƒ duyá»‡t. PhÃ²ng Ä‘Ã£ cÃ³ lá»‹ch Ä‘Æ°á»£c duyá»‡t trong khoáº£ng thá»i gian nÃ y.',
                    'error'
                );

                $this->redirect('staff_bookings');
            }
        }

        try {
            $this->deviceRepo->updateBookingStatus(
                $id,
                $status
            );

            if ($status === 'approved') {
                $this->setMessage(
                    'ÄÃ£ Ä‘á»“ng Ã½ yÃªu cáº§u Ä‘áº·t phÃ²ng!',
                    'success'
                );
            } else {
                $this->setMessage(
                    'ÄÃ£ tá»« chá»‘i yÃªu cáº§u Ä‘áº·t phÃ²ng!',
                    'success'
                );
            }
        } catch (PDOException $e) {
            $this->setMessage(
                'KhÃ´ng thá»ƒ cáº­p nháº­t yÃªu cáº§u Ä‘áº·t phÃ²ng.',
                'error'
            );
        }

        $this->redirect('staff_bookings');
    }

    /*
     * =====================================================
     * TV3 - Xá»¬ LÃ Äáº¶T PHÃ’NG
     * =====================================================
     */
    private function handleBooking(int $userId): array
    {
        $errors = [];
        $success = '';

        $roomId = trim(
            (string) ($_POST['room_id'] ?? '')
        );

        $date = trim(
            (string) ($_POST['date'] ?? '')
        );

        $start = trim(
            (string) ($_POST['start'] ?? '')
        );

        $end = trim(
            (string) ($_POST['end'] ?? '')
        );

        $purpose = trim(
            (string) ($_POST['purpose'] ?? '')
        );


        /*
         * VALIDATE
         */

        if ($roomId === '') {
            $errors['room_id']
                = 'Vui lÃ²ng chá»n phÃ²ng.';
        }

        if ($date === '') {
            $errors['date']
                = 'Vui lÃ²ng chá»n ngÃ y.';
        }

        if ($start === '') {
            $errors['start']
                = 'Vui lÃ²ng chá»n giá» báº¯t Ä‘áº§u.';
        }

        if ($end === '') {
            $errors['end']
                = 'Vui lÃ²ng chá»n giá» káº¿t thÃºc.';
        }

        if (
            $start !== ''
            && $end !== ''
            && $end <= $start
        ) {
            $errors['end']
                = 'Giá» káº¿t thÃºc pháº£i lá»›n hÆ¡n giá» báº¯t Ä‘áº§u.';
        }

        if ($purpose === '') {
            $errors['purpose']
                = 'Vui lÃ²ng nháº­p má»¥c Ä‘Ã­ch.';
        }


        if (!empty($errors)) {
            return [$errors, $success];
        }


        /*
         * KIá»‚M TRA PHÃ’NG
         */

        $room = $this->roomRepo->findBookableRoom(
            (int) $roomId
        );

        if (!$room) {

            $errors['room_id']
                = 'PhÃ²ng khÃ´ng tá»“n táº¡i hoáº·c Ä‘ang báº£o trÃ¬.';

            return [$errors, $success];
        }


        /*
         * Táº O DATETIME
         */

        $startTime =
            $date . ' ' . $start . ':00';

        $endTime =
            $date . ' ' . $end . ':00';


        /*
         * KIá»‚M TRA TRÃ™NG Lá»ŠCH
         *
         * TV3
         */

        if (
            $this->roomRepo->hasBookingConflict(
                (int) $roomId,
                $startTime,
                $endTime
            )
        ) {

            $errors['room_id']
                = 'PhÃ²ng Ä‘Ã£ Ä‘Æ°á»£c Ä‘áº·t trong thá»i gian nÃ y.';

            return [$errors, $success];
        }


        /*
         * INSERT BOOKING
         */

        $this->roomRepo->createBooking(
            $userId,
            (int) $roomId,
            $startTime,
            $endTime,
            $purpose
        );

        $success = 'Äáº·t phÃ²ng thÃ nh cÃ´ng.';

        return [$errors, $success];
    }


    /*
     * =====================================================
     * TV3 - Há»¦Y BOOKING
     * =====================================================
     */
    private function handleCancelBooking(
        int $userId
    ): array {

        $errors = [];
        $success = '';

        $bookingId =
            (int) ($_POST['booking_id'] ?? 0);


        if ($bookingId <= 0) {

            $errors['cancel']
                = 'YÃªu cáº§u Ä‘áº·t phÃ²ng khÃ´ng há»£p lá»‡.';

            return [$errors, $success];
        }


        /*
         * CHá»ˆ ÄÆ¯á»¢C Há»¦Y BOOKING Cá»¦A CHÃNH MÃŒNH
         */

        $booking =
            $this->roomRepo->findBookingForUser(
                $bookingId,
                $userId
            );


        if (!$booking) {

            $errors['cancel']
                = 'KhÃ´ng tÃ¬m tháº¥y yÃªu cáº§u Ä‘áº·t phÃ²ng cá»§a báº¡n.';

            return [$errors, $success];
        }


        /*
         * KIá»‚M TRA CÃ“ THá»‚ Há»¦Y KHÃ”NG
         */

        $canCancel =
            $booking['status'] !== 'cancelled'
            &&
            (
                $booking['status'] === 'pending'
                ||
                $booking['start_time']
                    > date('Y-m-d H:i:s')
            );


        if (!$canCancel) {

            $errors['cancel']
                = 'KhÃ´ng thá»ƒ há»§y yÃªu cáº§u nÃ y.';

            return [$errors, $success];
        }


        $this->roomRepo->cancelBooking(
            $bookingId,
            $userId
        );

        $success =
            'Há»§y yÃªu cáº§u Ä‘áº·t phÃ²ng thÃ nh cÃ´ng.';

        return [$errors, $success];
    }

     */
    private function handleReport(

    ): array {

        $errors = [];

        $deviceId =
            trim(
                (string) ($_POST['device_id'] ?? '')
            );

        $description =

        if (

        ) {

                = 'Thiáº¿t bá»‹ khÃ´ng tá»“n táº¡i.';

        /*

        );

        $success =
            'BÃ¡o há»ng thiáº¿t bá»‹ thÃ nh cÃ´ng.';

                 ORDER BY id DESC
                 LIMIT 1"
            );

            $stmt->execute([$deviceId]);

                throw new Exception(
                    'Thiáº¿t bá»‹ nÃ y Ä‘ang cÃ³ phiáº¿u báº£o trÃ¬.'

            $stmt = $this->pdo->prepare(

                    status,

                )
                VALUES
                (?, ?, ?, 'Äang báº£o trÃ¬', NOW())"
            );

            $stmt->execute([
                $deviceId,
                $staffId,
                $description
            ]);

            $maintenanceId = (int)$this->pdo->lastInsertId();

            $stmt = $this->pdo->prepare(
                "UPDATE devices
                 SET status = 'maintenance'
                 WHERE id = ?"
            );

            $stmt->execute([$deviceId]);

            $stmt = $this->pdo->prepare(
                "INSERT INTO maintenance_history
                (
                    maintenance_id,
                    device_id,
                    staff_id,
                    maintenance_date,

                $deviceId,
                $staffId,

                ]);

function dateKey(value){

    const match = String(value).match(
        /(\d{2})\/(\d{2})\/(\d{4})/
    );

    if(match){
        return `${match[3]}-${match[2]}-${match[1]}`;
    }

    const date = new Date(value);

    if(!isNaN(date.getTime())){
        return `${date.getFullYear()}-${String(
            date.getMonth()+1
        ).padStart(2,'0')}-${String(
            date.getDate()
        ).padStart(2,'0')}`;
    }

    return "";
}

function findRoom(id){

    const item = calendarRooms.find(

}

function taoLichCalendar(){

    const grid =
        document.getElementById("calendarGrid");

    const title =
        document.getElementById("calendarTitle");

    if(!grid || !title){
        return;
    }

    const year =
        calendarDate.getFullYear();

    const month =
        calendarDate.getMonth();

    title.textContent =
        `ThÃ¡ng ${month + 1} ${year}`;

    const names = [
        "T2",
        "T3",
        "T4",
        "T5",
        "T6",
        "T7",
        "CN"
    ];

    let html = names
        .map(
            x => `<div class="calendar-weekday">${x}</div>`
        )
        .join("");

    let start =
        new Date(year, month, 1).getDay();

    start =
        start === 0
        ? 6
        : start - 1;

    const days =
        new Date(year, month + 1, 0).getDate();

    const prevDays =
        new Date(year, month, 0).getDate();

    const total =
        Math.ceil((start + days) / 7) * 7;

    const now =
        new Date();

    const today =
        `${now.getFullYear()}-${String(
            now.getMonth()+1
        ).padStart(2,"0")}-${String(
            now.getDate()
        ).padStart(2,"0")}`;

    for(let i = 0; i < total; i++){

        let day;
        let cell;

        if(i < start){

            day =
                prevDays - start + i + 1;

            cell =
                new Date(
                    year,
                    month - 1,
                    day
                );

        }else if(i < start + days){

            day =
                i - start + 1;

            cell =
                new Date(
                    year,
                    month,
                    day
                );

        }else{

            day =
                i - start - days + 1;

            cell =
                new Date(
                    year,
                    month + 1,
                    day
                );
        }

        const key =
            `${cell.getFullYear()}-${String(
                cell.getMonth()+1
            ).padStart(2,"0")}-${String(
                cell.getDate()
            ).padStart(2,"0")}`;

        let events = [];

                        }`
                });
            }
        });

        borrowingCalendarData.forEach(item => {

            if(
                item.status !== "rejected" &&
                dateKey(item.start_time) === key
            ){

                events.push({

                    type:"borrow",

                    text:
                        `${String(item.start_time).split(" ")[0]} ${
                            findDevice(item.device_id)

                    detail:
                        `${item.user_name || ""} â€¢ ${
                            item.start_time
                        } - ${
                            item.end_time
                        }`
                });
            }
        });

        const visible =
            events.slice(0,3);

        const hidden =
            events.length - visible.length;

        const other =
            cell.getMonth() !== month
            ? "other-month"
            : "";

        const current =
            key === today

            : "";

        html +=

use App\Repository\DeviceRepository;
use App\Controller\DeviceController;
use App\Repository\MaintenanceRepository;
use App\Controller\MaintenanceController;

$route = $_GET['route'] ?? $_GET['page'] ?? 'login';

$pdo = Database::connection();

switch ($route) {
    case 'login':
        require_once __DIR__ . '/../src/View/auth/login.php';
        break;

    case 'register':
        require_once __DIR__ . '/../src/View/auth/register.php';
        break;

    case 'logout':
        session_unset();
        session_destroy();
        header('Location: index.php?route=login');
        exit;

    case 'users':
        $userRepo = new UserRepository($pdo);
        $userController = new UserController($userRepo);
        $userController->index();
        break;

    case 'bookings':
    case 'rooms':
        $bookingRepo = new RoomBookingRepository($pdo);
        $bookingController = new RoomBookingController($bookingRepo);
        $bookingController->index();
        break;

    // LUá»’NG Má»šI Cá»¦A AN: QUáº¢N LÃ THIáº¾T Bá»Š & DUYá»†T Lá»ŠCH
    case 'devices':
    case 'staff_bookings':
    case 'borrowings':
        $deviceRepo = new DeviceRepository($pdo);
        $deviceController = new DeviceController($deviceRepo);
        
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $deviceController->handleRequest();
        } else {
            $deviceController->index();
        }
        break;

    // LUá»’NG Má»šI Cá»¦A AN: Báº¢O TRÃŒ
    case 'maintenance':
        $maintenanceRepo = new MaintenanceRepository($pdo);
        $maintenanceController = new MaintenanceController($maintenanceRepo);

        break;

    default:
        http_response_code(404);
        echo "404 - KhÃ´ng tÃ¬m tháº¥y trang yÃªu cáº§u!";
        break;
